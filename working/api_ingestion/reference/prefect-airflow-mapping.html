CLI Cheat Sheet

# Install
pip install prefect

# Start local server (HQ dashboard)
prefect server start

# Run flow trực tiếp
python prefect_flow.py --endpoint entity_query --env dev

# Dry run (validate only)
python prefect_flow.py --endpoint entity_query --dry-run

# Deploy with schedule
prefect deploy prefect_flow.py:api_ingestion_pipeline \
    --name "clarizen-daily" \
    --cron "0 2 * * *"

# Xem deployments
prefect deployment ls

# Trigger manual run từ CLI
prefect deployment run api_ingestion_pipeline/clarizen-daily

# Xem flow runs
prefect flow-run ls
